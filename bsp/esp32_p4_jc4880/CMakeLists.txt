cmake_minimum_required(VERSION 3.16)

set(EXTRA_COMPONENT_DIRS 
    "${CMAKE_CURRENT_LIST_DIR}/../../core"
)

# Aislamiento estricto de perfiles por carpeta de compilación (-B build/<perfil> o -B build_<perfil>)
if(CMAKE_BINARY_DIR MATCHES "(/red|build_red)$")
    message(STATUS "[CBDos-P4] Perfil: CYBERDECK RED (Aislado en ${CMAKE_BINARY_DIR}/sdkconfig)")
    set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.red")
elseif(CMAKE_BINARY_DIR MATCHES "(/lite|build_lite)$")
    message(STATUS "[CBDos-P4] Perfil: CYBERDECK LITE (Aislado en ${CMAKE_BINARY_DIR}/sdkconfig)")
    set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.lite")
elseif(CMAKE_BINARY_DIR MATCHES "(/kds|build_kds)$")
    message(STATUS "[CBDos-P4] Perfil: TABLEHUB KDS COCINA (Aislado en ${CMAKE_BINARY_DIR}/sdkconfig)")
    set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.kds")
elseif(CMAKE_BINARY_DIR MATCHES "(/tabletop|build_tabletop)$")
    message(STATUS "[CBDos-P4] Perfil: TABLEHUB TABLETOP MESA (Aislado en ${CMAKE_BINARY_DIR}/sdkconfig)")
    set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.tabletop")
else()
    message(STATUS "[CBDos-P4] Perfil: CYBERDECK FULL (Maestro en sdkconfig)")
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults" CACHE STRING "" FORCE)
endif()

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(cbdos_p4)
